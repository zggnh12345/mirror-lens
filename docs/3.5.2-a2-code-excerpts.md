```go
// claim 返回 false 表示同一 (source, host) 正在跑或还在退避窗口内。
func (r *HostProbeRunner) claim(key string) bool {
	r.mu.Lock()
	defer r.mu.Unlock()
	st := r.state[key]
	now := r.o.Now()
	if st != nil {
		if st.running || st.next.After(now) {
			return false
		}
		delete(r.state, key)
	}
	r.state[key] = &probeState{next: now, running: true}
	return true
}

// noteSnapshot 记住本次探测使用的凭据版本。版本变化时把退避重置，
// 否则凭据早已稳定，仍会被旧版本的 10 分钟退避压着。
func (r *HostProbeRunner) noteSnapshot(key string, gen int64, digest string) {
	r.mu.Lock()
	defer r.mu.Unlock()
	st := r.state[key]
	if st == nil {
		return
	}
	if st.gen != gen || st.digest != digest {
		st.backoff = 0
		st.gen, st.digest = gen, digest
	}
}

func (r *HostProbeRunner) finish(key string, retry bool) {
	r.mu.Lock()
	defer r.mu.Unlock()
	st := r.state[key]
	if !retry {
		delete(r.state, key)
		return
	}
	prev := r.o.Backoff
	var gen int64
	var digest string
	if st != nil {
		gen, digest = st.gen, st.digest
		if st.backoff > 0 {
			prev = st.backoff * 2
		}
	}
	if prev > maxHostProbeBackoff {
		prev = maxHostProbeBackoff
	}
	r.state[key] = &probeState{next: r.o.Now().Add(prev), backoff: prev, gen: gen, digest: digest}
}
```

```go
	if err != nil {
		switch {

			// 不能靠重试掩盖。

			retry = true

			retry = true
		}
		return
	}

				Password:    snap.Credential.Password,
				TokenMethod: snap.Credential.TokenMethod,
				TokenAuth:   snap.Credential.TokenAuth,
			},
		},
		Realms:  r.o.Realms,
		Timeout: r.o.Timeout,
	})

	capv, mode, note, err := out.StoreFields()
	if err != nil {
		log.Printf("Host %s 探测结果自相矛盾，拒绝写库: %v", j.Host, err)
		return
	}
	// 一次事务写回 cap / scoped 健康 / recovery。gen 与 digest 都来自
	// 探测起点的快照，事务内只做相等校验，不重新读取当前代次。
	res, err := r.o.Store.CommitHostProbe(store.HostProbeCommit{
		SourceID:      j.SourceID,
		Host:          j.Host,
		Cap:           capv,
		PathMode:      mode,
		Reason:        out.Reason,
		Transport:     out.Transport,
		At:            r.o.Now().Unix(),
		ErrText:       note,
		Gen:           snap.Generation,
		CredDigest:    snap.Digest,
		HasCredential: snap.HasCredential,
		TTL:           r.o.RecoveryTTL,
	})
	if err != nil {
		// 写库失败 = 本次结果无法确认：屏蔽该源直到下一次成功提交。
		// failClosed 只补充这一种保守隔离，不参与六态派生。
		log.Printf("写回 %s 能力失败: %v", j.Host, err)
		r.blockRoute(j, snap.Generation)
		r.invalidateRoutes()
		retry = true
		return
	}
	switch res {
	case store.HostCapsCommitted:
		// 成功提交也可能改变资格（ok→unsupported、路径模式变化等），
		// 先失效原始候选缓存，再解除本次对应的屏蔽。
		r.invalidateRoutes()
		r.unblockRoute(j)
		if out.Cap == probe.CapYes {
			log.Printf("Host %s 验证通过（%s）", j.Host, out.Selected)
		}
	case store.HostCapsStaleSnapshot:
		// 整笔拒绝：cap、scoped、recovery 都没有写入。退避后重新取快照。
		retry = true
	case store.HostCapsSourceChanged:
		// 源已换 Host：下一轮按新归属重新入队。
	case store.HostCapsSourceUnavailable:
		// 归档或删除：不再重排。
	}
}
```

### internal/probe/registryhost.go:559-587 — StoreFields 与 ErrOutcomeInconsistent
```go
// ErrOutcomeInconsistent 表示聚合结果自相矛盾，拒绝写库。
var ErrOutcomeInconsistent = errors.New("probe outcome 与 selected 不一致")

// StoreFields 把探测结论转成写库参数，并在这一层校验聚合语义：
// cap=1 必须指向一条自身 cap=1 的路径，其余情况不得带路径。
func (o RegistryHostOutcome) StoreFields() (cap int, pathMode string, note string, err error) {
	note = o.DiagnosticJSON()
	switch o.Cap {
	case capYes:
		if o.Selected != modeNative && o.Selected != modeHostPrefix {
			return 0, "", note, ErrOutcomeInconsistent
		}
		sel := o.Native
		if o.Selected == modeHostPrefix {
			sel = o.HostPrefix
		}
		if sel.Cap != capYes {
			return 0, "", note, ErrOutcomeInconsistent
		}
		return capYes, o.Selected, note, nil
	case capNo, capMismatch, capUnknown:
		if o.Selected != "" {
			return 0, "", note, ErrOutcomeInconsistent
		}
		return o.Cap, "", note, nil
	default:
		return 0, "", note, fmt.Errorf("未知 registry_cap=%d", o.Cap)
	}
}
```
